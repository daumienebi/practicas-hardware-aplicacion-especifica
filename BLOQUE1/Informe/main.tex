	\begin{figure}[H]
		\centering
		\IfFileExists{captura_p#1_#2.png}{%
			\includegraphics[width=0.85\textwidth]{captura_p#1_#2.png}%
		}{%
			\fbox{%
				\parbox[c][4.5cm][c]{0.80\textwidth}{%
					\centering
					Insertar aquí la captura\\[0.3cm]
					\texttt{captura\_p#1\_#2.png}%
				}%
			}%
		}
		\caption{#3}
		\label{fig:p#1-#2}
	\end{figure}%
}
